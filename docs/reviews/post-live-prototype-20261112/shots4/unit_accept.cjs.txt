async function openCold(browser, label, ctxOpts, theme, link = LINK) {
  const ctx = await browser.newContext({ locale: 'he-IL', ...ctxOpts });
  const p = await ctx.newPage();
  const errs = [];
  p.on('pageerror', (e) => errs.push(e.message.slice(0, 140)));
  await p.addInitScript((t) => { try { localStorage.setItem('sod2029:theme-preset', t); localStorage.setItem('sod2029:theme-preset-seeded', '1'); } catch (e) {} }, theme);
  const t0 = Date.now();
  await p.goto(link, { waitUntil: 'domcontentloaded', timeout: 90000 });
  await p.waitForSelector(`[id="${UNIT}"]`, { timeout: 60000 });
  await p.waitForLoadState('networkidle', { timeout: 40000 }).catch(() => {});
  await p.waitForTimeout(2500);
  const m = await p.evaluate((U) => {
    const el = document.getElementById(U), r = el.getBoundingClientRect();
    return {
      ua: navigator.userAgent.slice(0, 60), localStorageKeysAtStart: Object.keys(localStorage).filter((k) => k.includes('unit-prefs')).length,
      unitTopVisible: r.top > -5 && r.top < innerHeight * 0.5, top: Math.round(r.top),
      h2: (el.querySelector('h2') || {}).textContent, coins: el.querySelectorAll('[data-gem]').length,
      groups: [...el.querySelectorAll('.sod29-pu-group')].map((g) => g.dataset.sourceIdentity),
      bar: (el.querySelector('[data-pu-bar]') || {}).dataset ? el.querySelector('[data-pu-bar]').dataset.puBar : null,
      shareUrl: (el.querySelector('[data-pu-share-url]') || {}).dataset ? el.querySelector('[data-pu-share-url]').dataset.puShareUrl : null,
      images: el.querySelectorAll('.sod29-pu-image').length, imagesImg: el.querySelectorAll('img').length,
      sourceUnits: el.querySelectorAll('.sod29-source-unit').length,
      ovX: document.documentElement.scrollWidth - innerWidth,
      dupIds: (() => { const ids = [...document.querySelectorAll('[id]')].map((n) => n.id); return ids.filter((x, i) => ids.indexOf(x) !== i).length; })(),
    };
  }, UNIT);
  rec(`${label}:${theme}`, { ...m, ms: Date.now() - t0, errs });
  return { ctx, p };
}
